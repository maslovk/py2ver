// -----------------------------------------------------------------------------
// jtag_transceiver.sv: uart_transceiver's words over the JTAG UART (jtag_bytes)
// instead of the UART's pins: a word in of RX_BYTES bytes, least significant
// first (rx_valid a clock when the last is in); tx_data's TX_BYTES bytes out
// the same way from the clock tx_valid is high (tx_busy until all are handed on).
// The next word's bytes are taken once the reply to the last has been handed on
// (they wait in the FIFO and the PC's till then): the PC may send many at once.
// -----------------------------------------------------------------------------
module jtag_transceiver #(
    parameter int TX_WIDTH = 32,
    parameter int RX_WIDTH = 32
)(
    input  logic                clk,
    input  logic                rst_n,
    input  logic [TX_WIDTH-1:0] tx_data,
    input  logic                tx_valid,
    output logic                tx_busy,
    output logic [RX_WIDTH-1:0] rx_data,
    output logic                rx_valid
);
    localparam int TX_BYTES = (TX_WIDTH + 7) / 8;
    localparam int RX_BYTES = (RX_WIDTH + 7) / 8;

    logic [7:0] rx_byte, tx_byte;
    logic       rx_byte_ready, tx_start, tx_busy_b;
    logic       waiting;    // (a word in, its reply not yet handed on)

    jtag_bytes u_bytes (
        .clk, .rst_n,
        .rx_byte, .rx_byte_ready, .rx_accept(!waiting),
        .tx_byte, .tx_start, .tx_busy(tx_busy_b)
    );

    logic [TX_WIDTH-1:0]           tx_shift;
    logic [$clog2(TX_BYTES+1)-1:0] tx_left;
    logic [$clog2(RX_BYTES):0]     rx_idx;
    logic [RX_WIDTH-1:0]           rx_accum, next_accum;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            tx_busy  <= 1'b0;
            tx_start <= 1'b0;
            tx_byte  <= 8'h00;
            tx_shift <= '0;
            tx_left  <= '0;
            rx_idx   <= '0;
            rx_accum <= '0;
            rx_data  <= '0;
            rx_valid <= 1'b0;
            waiting  <= 1'b0;
        end else begin
            tx_start <= 1'b0;
            rx_valid <= 1'b0;

            // ------------------------------ TX ------------------------------
            if (!tx_busy) begin
                if (tx_valid) begin
                    tx_shift <= tx_data;
                    tx_left  <= TX_BYTES;
                    tx_busy  <= 1'b1;
                end
            end else if (!tx_start && !tx_busy_b) begin
                if (tx_left != '0) begin
                    tx_byte  <= tx_shift[7:0];
                    tx_start <= 1'b1;
                    tx_shift <= tx_shift >> 8;
                    tx_left  <= tx_left - 1'b1;
                end else begin
                    tx_busy  <= 1'b0;
                    waiting  <= 1'b0;
                end
            end

            // ------------------------------ RX ------------------------------
            if (rx_byte_ready) begin
                next_accum = rx_accum;
                next_accum[(rx_idx*8) +: 8] = rx_byte;
                if (rx_idx + 1 < RX_BYTES) begin
                    rx_accum <= next_accum;
                    rx_idx   <= rx_idx + 1'b1;
                end else begin
                    rx_accum <= '0;
                    rx_data  <= next_accum;
                    rx_valid <= 1'b1;
                    rx_idx   <= '0;
                    waiting  <= 1'b1;
                end
            end
        end
    end
endmodule
