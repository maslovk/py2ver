// Core handshake for the UART board wrapper.
//
// After input_valid (the UART has new, now-held core inputs), wait LATENCY
// clocks for the core's result, latch every output once, then pulse
// `ready` (send the outputs) and `step` (advance a stateful core by one
// sample, after its outputs were captured). With PRE_STEP = 1 (a core with
// block RAM), `step` is also pulsed right after input_valid, for the clock
// in which the core reads its RAM. With USE_VOUT = 1 (a pipelined core with
// state, which tracks its samples itself), the outputs are latched when the
// core raises vout instead of after LATENCY clocks.
module delayed_registers #(
    parameter int LATENCY = 0,
    parameter int PRE_STEP = 0,
    parameter int USE_VOUT = 0
) (
    input  logic clk,
    input  logic rst_n, // Active-low reset
    input  logic input_valid,
    input  logic vout,
{%- for output in outputs %}
    input  logic [{{ output.width - 1 }}:0] {{ output.name }},
    output logic [{{ output.width - 1 }}:0] {{ output.name }}_out,
{%- endfor %}
    output logic ready,
    output logic step
);

    logic busy;
    logic [$clog2(LATENCY + PRE_STEP + 2) - 1:0] count;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            busy  <= 1'b0;
            count <= '0;
            ready <= 1'b0;
            step  <= 1'b0;
{%- for output in outputs %}
            {{ output.name }}_out <= '0;
{%- endfor %}
        end else begin
            ready <= 1'b0;
            step  <= 1'b0;
            if (input_valid) begin
                busy  <= 1'b1;
                count <= '0;
                step  <= (PRE_STEP != 0);
            end else if (busy) begin
                if (USE_VOUT != 0 ? vout : (count == LATENCY + PRE_STEP)) begin
{%- for output in outputs %}
                    {{ output.name }}_out <= {{ output.name }};
{%- endfor %}
                    ready <= 1'b1;
                    step  <= 1'b1;
                    busy  <= 1'b0;
                end else begin
                    count <= count + 1'b1;
                end
            end
        end
    end

endmodule
